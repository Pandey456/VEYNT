        .getElementById("connectBtn")

          signer = { address: accounts[0] };

            abi: ABI,
            client: {
              public: provider,
              wallet: provider,
            },
          });

          console.log("Frontend Contract Address:", CONTRACT_ADDRESS);

          console.log("Contract Owner:", await contract.read.owner());

            "Target USD Fee:",
            (await contract.read.TARGET_USD_FEE()).toString(),
          );

          const code = await provider.getCode({
            address: CONTRACT_ADDRESS,
          });

          console.log("Contract code:", code);

          document.getElementById("walletStatus").innerText =
            `Connected: ${signer.address}`;

        .getElementById("createMarketBtn")
        .addEventListener("click", async () => {
          if (!contract) return alert("Connect wallet first.");
          try {
            const question = document.getElementById("question").value;
            const endpoint = document.getElementById("endpoint").value;
            const minutes = parseInt(
              document.getElementById("deadlineOffset").value,
            );

            const targetMs = Date.now() + minutes * 60 * 1000;
            const deadlineSec = Math.floor(targetMs / 60000) * 60;
            //Debug
            console.log("FTSO Address:", await contract.read.i_ftsoV2());

            console.log("Feed ID:", await contract.read.i_flrUsdFeedId());

            console.log("TEE Signer:", await contract.read.i_teeSigner());

            //end debug

            console.log("Fetching FTSO fee...");

            //const fee = await contract.read.getRequiredFee();
            const feeweGOt = await contract.read.getRequiredFee();

            const fee = feeweGOt + (feeweGOt * 5n) / 100n;

            console.log("Fee just before tx:", fee.toString());

            await new Promise((r) => setTimeout(r, 5000));

            const feeAgain = await contract.read.getRequiredFee();

            console.log("Fee 5 seconds later:", feeAgain.toString());

            console.log("Sending Create Market TX...");
            const hash = await contract.write.createMarket(
              [question, BigInt(deadlineSec), endpoint],
              { value: fee, account: signer.address },
            );
            const tx = {
              hash,
              async wait() {
                return await provider.waitForTransactionReceipt({ hash });
              },
            };
            await tx.wait();
            alert(`Market created! TX: ${tx.hash}`);
          } catch (err) {
            console.error(err);
            alert("Failed: " + err.message);
          }

        .getElementById("finalizeBtn")

          const repo = document.getElementById("ghRepo").value.trim();
          const token = document.getElementById("ghToken").value.trim();
          const marketId = document.getElementById("finalizeMarketId").value;
          const statusEl = document.getElementById("finalizeStatus");

            return alert(
              "Please enter both GitHub Repo (owner/repo) and PAT Token.",
            );

          statusEl.innerText = "Triggering GitHub Actions workflow...";

          try {
            const response = await fetch(
              `https://api.github.com/repos/${repo}/actions/workflows/fdc-full-run.yml/dispatches`,
              {
                method: "POST",
                headers: {
                  Accept: "application/vnd.github.v3+json",
                  Authorization: `Bearer ${token}`,
                  "Content-Type": "application/json",
                },
                body: JSON.stringify({
                  ref: "main",
                  inputs: { market_id: marketId.toString() },
                }),
              },
            );

            if (response.status === 204) {
              statusEl.innerText =
                "✅ Workflow triggered successfully! Check your GitHub Actions tab for live logs.";
            } else {
              const errData = await response.json();
              statusEl.innerText = `❌ Error (${response.status}): ${errData.message}`;
            }
          } catch (err) {
            console.error(err);
            statusEl.innerText = "❌ Network error triggering workflow.";
          }

              proof = JSON.parse(proofInput);

              throw new Error(

              );

              },

              statusEl.innerText = `✅ Payout claimed successfully!\nTX: ${hash}`;

        .getElementById("withdrawTreasuryBtn")
        .addEventListener("click", async () => {
          if (!contract) {
            return alert("Connect wallet first.");
          }

          const statusEl = document.getElementById("adminStatus");

          try {
            const owner = await contract.read.owner();

            if (owner.toLowerCase() !== signer.address.toLowerCase()) {
              throw new Error(
                "Only the contract owner can withdraw treasury funds.",
              );
            }

            const to = document.getElementById("withdrawAddress").value.trim();

            if (!to) {
              throw new Error("Enter withdrawal address.");
            }

            statusEl.innerText = "Submitting treasury withdrawal...";

            const hash = await contract.write.withdrawTreasury([to], {
              account: signer.address,
            });

            statusEl.innerText = `Transaction submitted: ${hash}`;

            const receipt = await provider.waitForTransactionReceipt({
              hash,
            });

            if (receipt.status === "success") {
              statusEl.innerText = `✅ Treasury withdrawn successfully!\nTX: ${hash}`;
            } else {
              statusEl.innerText = "❌ Treasury withdrawal failed.";
            }
          } catch (err) {
            console.error(err);

            statusEl.innerText = `❌ Withdrawal failed: ${
              err.shortMessage || err.message
            }`;
          }
        });
    </script>
  </body>
</html>
