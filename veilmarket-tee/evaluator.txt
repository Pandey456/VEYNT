const {
  createPublicClient,
  createWalletClient,
  http,
  decodeAbiParameters,
  parseAbiParameters,
  keccak256,
  encodePacked,
  parseAbiItem,
} = require("viem");
const { privateKeyToAccount } = require("viem/accounts");
const { coston2 } = require("viem/chains");
const { StandardMerkleTree } = require("@openzeppelin/merkle-tree");

// Import the unmodified main function from your fdc-run.js
const { main: runFdc } = require("./fdc-run.js");

// --- CONFIGURATION ---
const RPC_URL =
  process.env.RPC_URL || "https://coston2-api.flare.network/ext/C/rpc";
const VEIL_MARKET_ADDRESS =
  process.env.VEIL_MARKET_ADDRESS ||
  "0x0000000000000000000000000000000000000000"; // Replace with deployed contract address
const MARKET_ID = BigInt(process.env.MARKET_ID || "1");

// Wallet Keys
if (!process.env.PRIVATE_KEY || !process.env.TEE_PRIVATE_KEY) {
  console.error(
    "ERROR: Please set PRIVATE_KEY and TEE_PRIVATE_KEY in your environment.",
  );
  process.exit(1);
}

const submitterAccount = privateKeyToAccount(process.env.PRIVATE_KEY);
const teeAccount = privateKeyToAccount(process.env.TEE_PRIVATE_KEY);

const publicClient = createPublicClient({
  chain: coston2,
  transport: http(RPC_URL),
});
const walletClient = createWalletClient({
  account: submitterAccount,
  chain: coston2,
  transport: http(RPC_URL),
});

// VeilMarket Contract ABI
const VEIL_MARKET_ABI = [
  {
    type: "function",
    name: "markets",
    stateMutability: "view",
    inputs: [{ type: "uint256", name: "marketId" }],
    outputs: [
      { type: "address", name: "owner" },
      { type: "string", name: "apiEndpoint" },
      { type: "string", name: "question" },
      { type: "uint256", name: "deadline" },
      { type: "uint256", name: "totalPool" },
      { type: "bool", name: "resolved" },
      { type: "bool", name: "outcome" },
      { type: "bytes32", name: "merkleRoot" },
    ],
  },
  {
    type: "function",
    name: "resolveMarket",
    stateMutability: "nonpayable",
    inputs: [
      { type: "uint256", name: "_marketId" },
      { type: "bytes32", name: "_merkleRoot" },
      { type: "string", name: "_outcome" },
      { type: "bytes", name: "_signature" },
    ],
    outputs: [],
  },
  {
    type: "event",
    name: "PredictionPlaced",
    inputs: [
      { type: "uint256", name: "marketId", indexed: true },
      { type: "address", name: "bettor", indexed: true },
      { type: "uint256", name: "amount", indexed: false },
      { type: "bytes", name: "encryptedChoice", indexed: false },
    ],
  },
];

/**
 * Main Evaluation Controller
 */
async function evaluate() {
  console.log(`\n==================================================`);
  console.log(`=== STARTING EVALUATION FOR MARKET #${MARKET_ID} ===`);
  console.log(`==================================================\n`);

  // --- STEP 1: FETCH MARKET DETAILS FROM SMART CONTRACT ---
  console.log("Step 1 — Reading market metadata from VeilMarket.sol...");
  const market = await publicClient.readContract({
    address: VEIL_MARKET_ADDRESS,
    abi: VEIL_MARKET_ABI,
    functionName: "markets",
    args: [MARKET_ID],
  });

  if (market.owner === "0x0000000000000000000000000000000000000000") {
    throw new Error(`Market #${MARKET_ID} does not exist on contract.`);
  }
  if (market.resolved) {
    throw new Error(`Market #${MARKET_ID} is already resolved.`);
  }

  // Parse market parameters from question string (Format: "BTC|ABOVE|6500000000000")
  let direction = "ABOVE";
  let targetPrice = 6500000000000n; // $65,000 scaled by 10^8

  if (market.question.includes("|")) {
    const parts = market.question.split("|");
    direction = parts[1].toUpperCase();
    targetPrice = BigInt(parts[2]);
  }

  console.log(`Question:    "${market.question}"`);
  console.log(
    `Condition:   Price must be ${direction} $${Number(targetPrice) / 1e8}`,
  );
  console.log(
    `Deadline:    ${new Date(Number(market.deadline) * 1000).toISOString()}`,
  );

  // --- STEP 2: EXECUTE FDC ATTESTATION VIA FDC-RUN.JS ---
  console.log("\nStep 2 — Triggering FDC proof generation via fdc-run.js...");
  const proofData = await runFdc();

  if (!proofData || !proofData.response || !proofData.response.responseBody) {
    throw new Error("FDC execution failed to return valid proof data.");
  }

  const fdcAbiEncodedData = proofData.response.responseBody.abiEncodedData;
  console.log("Verified FDC Hex Data:", fdcAbiEncodedData);

  // --- STEP 3: DECODE VERIFIED FDC PRICE ---
  const [{ price: verifiedPrice }] = decodeAbiParameters(
    parseAbiParameters("(uint256 price)"),
    fdcAbiEncodedData,
  );

  console.log(`\nStep 3 — Decoded FDC Price: $${Number(verifiedPrice) / 1e8}`);

  // --- STEP 4: MATCH FDC PRICE AGAINST MARKET CONDITION ---
  let winningOutcome;
  if (direction === "ABOVE") {
    winningOutcome = verifiedPrice >= targetPrice ? "YES" : "NO";
  } else if (direction === "BELOW") {
    winningOutcome = verifiedPrice <= targetPrice ? "YES" : "NO";
  } else {
    throw new Error(`Unsupported direction: ${direction}`);
  }

  console.log(`Winning Outcome: [ ${winningOutcome} ]`);

  // --- STEP 5: DECRYPT BETS & CALCULATE 86% POOL PAYOUTS ---
  console.log("\nStep 4 — Fetching prediction events from contract...");
  const logs = await publicClient.getLogs({
    address: VEIL_MARKET_ADDRESS,
    event: parseAbiItem(
      "event PredictionPlaced(uint256 indexed marketId, address indexed bettor, uint256 amount, bytes encryptedChoice)",
    ),
    args: { marketId: MARKET_ID },
    fromBlock: 0n,
  });

  let totalPoolWei = 0n;
  let winningPoolWei = 0n;
  const winnerStakes = new Map(); // address -> total winning stake wei

  for (const log of logs) {
    const bettor = log.args.bettor;
    const amount = log.args.amount;

    // Decrypt vote (In hackathon test, choice is hex-encoded string e.g. "YES" or "NO")
    let choice = "YES";
    try {
      choice = Buffer.from(log.args.encryptedChoice.slice(2), "hex")
        .toString("utf8")
        .trim()
        .toUpperCase();
    } catch (e) {
      console.warn(
        `Could not decode vote for ${bettor}, assuming raw payload.`,
      );
    }

    totalPoolWei += amount;

    if (choice === winningOutcome) {
      winningPoolWei += amount;
      const currentStake = winnerStakes.get(bettor) || 0n;
      winnerStakes.set(bettor, currentStake + amount);
    }
  }

  console.log(`Total Bets Pool:    ${totalPoolWei.toString()} wei`);
  console.log(`Winning Bets Pool:  ${winningPoolWei.toString()} wei`);

  if (winningPoolWei === 0n) {
    console.warn("No winning predictions were placed on this market.");
    return;
  }

  // 86% Distributable Pool Formula: Payout = (UserBet / TotalWinningBets) * (TotalPool * 0.86)
  const distributablePoolWei = (totalPoolWei * 86n) / 100n;
  const merkleLeaves = [];

  console.log("\nPro-Rata Winners Breakdown (86% Pool):");
  for (const [bettor, userStake] of winnerStakes.entries()) {
    const payoutWei = (userStake * distributablePoolWei) / winningPoolWei;
    merkleLeaves.push([bettor, payoutWei.toString()]);
    console.log(
      ` Winner: ${bettor} -> Stake: ${userStake} wei | Payout: ${payoutWei} wei`,
    );
  }

  // --- STEP 6: CONSTRUCT MERKLE TREE ---
  console.log("\nStep 5 — Building OpenZeppelin Merkle Tree...");
  const tree = StandardMerkleTree.of(merkleLeaves, ["address", "uint256"]);
  const merkleRoot = tree.root;
  console.log("Merkle Root Generated:", merkleRoot);

  // --- STEP 7: SIGN RESOLUTION WITH TEE PRIVATE KEY ---
  // Must match contract verification hash: keccak256(abi.encodePacked(_marketId, _merkleRoot, _outcome))
  console.log("\nStep 6 — Signing Merkle Root with TEE Signer Key...");
  const messageHash = keccak256(
    encodePacked(
      ["uint256", "bytes32", "string"],
      [MARKET_ID, merkleRoot, winningOutcome],
    ),
  );

  const signature = await teeAccount.signMessage({
    message: { raw: messageHash },
  });

  console.log("Signed by TEE Address:", teeAccount.address);
  console.log("Signature:", signature);

  // --- STEP 8: SUBMIT RESOLVE TRANSACTION TO SMART CONTRACT ---
  console.log(
    "\nStep 7 — Submitting resolveMarket transaction to VeilMarket.sol...",
  );
  const txHash = await walletClient.writeContract({
    address: VEIL_MARKET_ADDRESS,
    abi: VEIL_MARKET_ABI,
    functionName: "resolveMarket",
    args: [MARKET_ID, merkleRoot, winningOutcome, signature],
  });

  console.log(`Submitted resolveMarket tx: ${txHash}`);
  const receipt = await publicClient.waitForTransactionReceipt({
    hash: txHash,
  });
  console.log(`Transaction confirmed in block #${receipt.blockNumber}!`);

  // Print Tree Dump for claims
  console.log("\n==================================================");
  console.log("=== SUCCESS! MARKET RESOLVED ON-CHAIN ===");
  console.log("==================================================");
  console.log("\nSave this Merkle Tree dump for claiming payouts in frontend:");
  console.log(JSON.stringify(tree.dump(), null, 2));
}

// Execute Evaluation Engine
evaluate().catch((error) => {
  console.error("\nFATAL EVALUATOR ERROR:", error);
  process.exit(1);
});
